/* ========================================================= 명단 관리 ========================================================= */ const MEMBER_SPREADSHEET_ID = "1yLvbjSa-tK_3lv16kJQDk5zKaltlrNBRO6SWtY0Cejk"; const MEMBER_SHEET_NAME = "명단"; /* ========================================================= 명단 가져오기 ========================================================= */ function getMemberList() { const ss = SpreadsheetApp.openById( MEMBER_SPREADSHEET_ID ); const sheet = ss.getSheetByName( MEMBER_SHEET_NAME ); if (!sheet) { throw new Error( "명단 시트를 찾을 수 없습니다." ); } const lastRow = sheet.getLastRow(); /* * 첫 행에 제목이 없거나 * 이름 데이터가 없는 경우 */ if (lastRow < 2) { return []; } const values = sheet .getRange( 2, 1, lastRow - 1, 3 ) .getDisplayValues(); return values .filter(function(row) { return String( row[0] ).trim() !== ""; }) .map(function(row) { return { name: String(row[0]).trim(), volunteer: String(row[1]).toUpperCase() === "TRUE", exhibition: String(row[2]).toUpperCase() === "TRUE" }; }); } /* ========================================================= 명단 저장 ========================================================= */ function saveMemberList(members) { const ss = SpreadsheetApp.openById( MEMBER_SPREADSHEET_ID ); const sheet = ss.getSheetByName( MEMBER_SHEET_NAME ); if (!sheet) { throw new Error( "명단 시트를 찾을 수 없습니다." ); } /* * 첫 행 제목 */ sheet .getRange(1, 1, 1, 3) .setValues([ [ "이름", "봉사자", "전시대" ] ]); /* * 기존 데이터 삭제 */ const lastRow = sheet.getLastRow(); if (lastRow > 1) { sheet .getRange( 2, 1, lastRow - 1, 3 ) .clearContent(); } /* * 저장할 데이터가 없는 경우 */ if ( !members || members.length === 0 ) { return { success: true, count: 0 }; } const values = members.map(function(member) { return [ String( member.name || "" ).trim(), member.volunteer === true, member.exhibition === true ]; }); sheet .getRange( 2, 1, values.length, 3 ) .setValues(values); return { success: true, count: values.length }; } /* ========================================================= 명단 API ========================================================= */ function doGet(e) { const action = e && e.parameter && e.parameter.action ? e.parameter.action : ""; if (action === "members") { return ContentService .createTextOutput( JSON.stringify( getMemberList() ) ) .setMimeType( ContentService.MimeType.JSON ); } return ContentService .createTextOutput( JSON.stringify({ success: true, message: "명단 API 정상 작동" }) ) .setMimeType( ContentService.MimeType.JSON ); } /* ========================================================= 명단 저장 API ========================================================= */ function doPost(e) { try { const data = JSON.parse( e.postData.contents ); if ( data.action === "saveMembers" ) { const result = saveMemberList( data.members ); return ContentService .createTextOutput( JSON.stringify(result) ) .setMimeType( ContentService.MimeType.JSON ); } throw new Error( "알 수 없는 요청입니다." ); } catch (error) { return ContentService .createTextOutput( JSON.stringify({ success: false, error: error.message }) ) .setMimeType( ContentService.MimeType.JSON ); } }
