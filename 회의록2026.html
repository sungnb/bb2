// =============================== // 회의록 가져오기 (Apps Script) // =============================== const API_URL = "https://script.google.com/macros/s/AKfycbxk62F4H4IfNhTT1v9pifW5mzIcSk-mKzgallaA_M7XNLEcgTM5kY3qL6yn0qCKJ9I/exec"; let meetingData = []; async function loadMeetingData() { try { const response = await fetch(API_URL); const result = await response.json(); // 전시대참여자.html 방식 if (!result.success) { throw new Error("데이터를 가져오지 못했습니다."); } meetingData = groupMeetings(result.data); renderMeetingList(); } catch (err) { console.error(err); document.getElementById("meetingList").innerHTML = `
회의록을 불러오지 못했습니다.
`; } } // =============================== // 번호별 회의 자동 묶기 // =============================== function groupMeetings(rows) { const meetings = []; let current = null; rows.forEach(row => { const no = String(row.number || "").trim(); // 1,2,3,4 ... 시작 번호 if (/^[0-9]+$/.test(no)) { if (current) meetings.push(current); current = { meetingNo: no, title: row.title || `제 ${no}번 회의`, date: row.date || "", items: [row] }; } else if ( current && no.startsWith(current.meetingNo + ".") ) { current.items.push(row); } }); if (current) meetings.push(current); return meetings; }
